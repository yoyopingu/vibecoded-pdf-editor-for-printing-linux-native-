# CopyShop PDF Suite — runtime dependencies
#
# Derived by scanning every `import` in main.py and tools/*.py, not from prose.
# Versions are the floors the code is known to work against (what it is
# developed on); newer releases are expected to be fine.
#
# NOTE: these are the *Python* dependencies only. The app also shells out to
# system binaries that pip cannot install — ghostscript, poppler and (for OCR)
# tesseract. See README.md.

PyQt6>=6.6          # the entire UI; QtCore, QtGui, QtWidgets, QtNetwork, QtPrintSupport
pypdf>=4.0          # page-level read/write: order, rotation, merge, split, save
pikepdf>=8.0        # lower-level PDF surgery (linearise, layers, colour profiles)
pypdfium2>=4.0      # page rasterising for the viewer, thumbnails and previews
Pillow>=10.0        # image handling behind the renderers and converters
img2pdf>=0.5        # image -> PDF without re-encoding
reportlab>=4.0      # generated content: page numbers, cut marks, imposition
pdf2image>=1.16     # PDF -> images for the OCR fallback (needs poppler)
